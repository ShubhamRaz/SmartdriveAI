   442→  tick(rawDt: number) {
   443→    const dt = Math.min(rawDt, 0.05);
   444→    this.clock += dt;
   445→    const s = this.settings;
   446→
   447→    // ---------- vision merge ----------
   448→    this.updateDriverAndDrowsiness(dt, s);
   449→
   450→    // ---------- mode machine ----------
   451→    switch (this.mode) {
   452→      case "SAFETY_CHECK":
   453→        this.tickSafetyCheck(dt);
   454→        break;
   455→      case "READY":
   456→      case "MANUAL":
   457→      case "WARNING":
   458→        if (this.autopilot) {
   459→          this.tickAutopilot(dt);
   460→        } else {
   461→          this.tickDriving(dt, s);
   462→        }
   463→        break;
   464→      case "AUTONOMOUS":
   465→        this.tickAutonomous(dt);
   466→        break;
   467→      case "STOPPING":
   468→        this.tickStopping(dt);
   469→        break;
   470→      case "EMERGENCY":
   471→        this.tickEmergency(dt);
   472→        break;
   473→      case "STOPPED":
   474→        this.kin.throttle = 0;
   475→        this.kin.brake = 0;
   476→        this.kin.steering = 0;
   477→        break;
   478→      case "IDLE":
   479→        break;
   480→    }
   481→
   482→    // ---------- safety evaluation (10 Hz) ----------
   483→    this.safetyPulse(dt, s);
   484→  }

   486→  /* --------------------------- sub-routines ----------------------------- */
   487→
   488→  private tickSafetyCheck(dt: number) {
   489→    this.safetyCheckTimer += dt;
   490→    if (!this.helmetCheckAnnounced && this.safetyCheckTimer > 0.5) {
   491→      this.helmetCheckAnnounced = true;
   492→      if (VEHICLE_RULES[this.vehicleType].requireHelmetToStart) {
   493→        this.emit(
   494→          this.helmet.state === "HELMET_DETECTED" ? "INFO" : "WARNING",
   495→          "AI VISION",
   496→          this.helmet.state === "HELMET_DETECTED"
   497→            ? "Helmet detected (simulated detector)"
   498→            : "Helmet NOT detected",
   499→        );
   500→      }
   501→      this.emit(
   502→        this.alcohol.detected ? "CRITICAL" : "INFO",
   503→        "AI VISION",
   504→        this.alcohol.detected
   505→          ? `Alcohol sensor reading ${this.alcohol.level} — above threshold`
   506→          : "Alcohol sensor reading 0 — within safe range",
   507→      );
   508→    }
   509→    if (this.safetyCheckTimer >= 1.6) {
   510→      // finalize checks
   511→      const blockedHelmet =
   512→        VEHICLE_RULES[this.vehicleType].requireHelmetToStart &&
   513→        this.helmet.state !== "HELMET_DETECTED";
   514→      const blockedAlcohol = this.alcohol.detected;
   515→      if (blockedHelmet || blockedAlcohol) {
   516→        this.mode = "IDLE";
   517→        this.engineOn = false;
   518→        this.startBlockedReason = blockedHelmet
   519→          ? "HELMET NOT DETECTED"
   520→          : "ALCOHOL DETECTED";
   521→        this.emit(
   522→          "CRITICAL",
   523→          "SAFETY ENGINE",
   524→          `START BLOCKED — ${this.startBlockedReason}`,
   525→        );
   526→        this.safetyStatus = "CRITICAL";
   527→        this.decision = {
   528→          severity: "CRITICAL",
   529→          reason: blockedHelmet
   530→            ? "Helmet interlock: no helmet detected"
   531→            : "Alcohol interlock: sensor above threshold",
   532→          recommendedAction: "BLOCK_START",
   533→          controlMode: "LOCKED",
   534→          evidence: [],
   535→          trigger: blockedHelmet ? "HELMET" : "ALCOHOL",
   536→        };
   537→        this.sessionEventTally.critical += 1;
   538→        this.prevSeverity = "CRITICAL";
   539→        playCue("BLOCKED");
   540→      } else {
   541→        this.mode = "READY";
   542→        this.engineOn = true;
   543→        this.hazardsOn = false;
   544→        this.sessionStartedAt = Date.now();
   545→        this.emit("SAFE", "SAFETY ENGINE", "Safety checks passed");
   546→        this.emit("INFO", "VEHICLE ENGINE", "Engine started — manual control available");
   547→      }
   548→    }
   549→  }
   550→
   551→  private tickDriving(dt: number, s: AppSettings) {
   552→    // manual / warning driving: gather inputs (keyboard channel + on-screen pad)
   553→    let throttle = this.kin.throttle;
   554→    let brake = this.kin.brake;
   555→    let steer = this.kin.steering;
   556→
   557→    if (this.demoAutoThrottle) {
   558→      const cruise = (s.cruiseSpeed[this.vehicleType] / 3.6) * 0.92;
   559→      throttle = this.kin.speed < cruise ? 1 : 0.05;
   560→    } else {
   561→      // merge on-screen pedal input (strongest input wins)
   562→      throttle = Math.max(throttle, this.pad.throttle);
   563→      brake = Math.max(brake, this.pad.brake);
   564→      steer = Math.max(-1, Math.min(1, steer + this.pad.steer));
   565→    }
   566→
   567→    const res = stepPhysics(
   568→      this.kin.speed,
   569→      this.kin.position,
   570→      this.kin.lateral,
   571→      this.kin.odometer,
   572→      VEHICLE_SPECS[this.vehicleType],
   573→      { throttle, brake, steer, powerLimit: this.powerLimit },
   574→      dt,
   575→      null,
   576→    );
   577→    this.applyPhysics(res);
   578→
   579→    // mode semantics
   580→    if (this.decision.severity === "WARNING") {
   581→      this.mode = "WARNING";
   582→    } else {
   583→      this.mode = this.kin.speed > 0.5 ? "MANUAL" : "READY";
   584→    }
   585→
   586→    if (this.manualStopRequested) {
   587→      this.manualStopRequested = false;
   588→      this.beginControlledStop("MANUAL_STOP");
   589→    }
   590→  }

   592→  /** autopilot chauffeur: cruise control + lane keeping toward lane center */
   593→  private tickAutopilot(dt: number) {
   594→    // driver brake input disengages autopilot (same convention as real ADAS)
   595→    const brakeInput = Math.max(this.kin.brake, this.pad.brake);
   596→    if (brakeInput > 0.2) {
   597→      this.disengageAutopilot("driver brake input");
   598→      this.tickDriving(dt, this.settings);
   599→      return;
   600→    }
   601→
   602→    const cruise = this.settings.cruiseSpeed[this.vehicleType] / 3.6;
   603→    const err = cruise - this.kin.speed;
   604→    const throttle = err > 0.3 ? Math.min(1, 0.25 + err * 0.6) : err < -0.6 ? 0 : 0.08;
   605→    const res = stepPhysics(
   606→      this.kin.speed,
   607→      this.kin.position,
   608→      this.kin.lateral,
   609→      this.kin.odometer,
   610→      VEHICLE_SPECS[this.vehicleType],
   611→      { throttle, brake: 0, steer: 0, powerLimit: this.powerLimit },
   612→      dt,
   613→      0, // lane keeping: steer back to lane center
   614→    );
   615→    this.applyPhysics(res);
   616→
   617→    // mode semantics
   618→    this.mode =
   619→      this.decision.severity === "WARNING"
   620→        ? "WARNING"
   621→        : this.kin.speed > 0.5
   622→          ? "MANUAL"
   623→          : "READY";
   624→
   625→    if (this.manualStopRequested) {
   626→      this.manualStopRequested = false;
   627→      this.disengageAutopilot("operator stop request");
   628→      this.beginControlledStop("MANUAL_STOP");
   629→    }
   630→  }
   631→
   632→  private tickAutonomous(dt: number) {
   633→    const { cmd, done, phaseChanged } = autonomyTick(
   634→      this.autonomy,
   635→      this.kin.speed,
   636→      this.kin.lateral,
   637→      this.kin.position,
   638→      VEHICLE_SPECS[this.vehicleType],
   639→      dt,
   640→    );
   641→    if (phaseChanged) this.onAutonomyPhase(phaseChanged);
   642→
   643→    const res = stepPhysics(
   644→      this.kin.speed,
   645→      this.kin.position,
   646→      this.kin.lateral,
   647→      this.kin.odometer,
   648→      VEHICLE_SPECS[this.vehicleType],
   649→      {
   650→        throttle: cmd.throttle,
   651→        brake: cmd.brake,
   652→        steer: 0,
   653→        powerLimit: 1,
   654→      },
   655→      dt,
   656→      cmd.lateralTarget,
   657→    );
   658→    this.applyPhysics(res);
   659→
   660→    if (done) {
   661→      this.finalizeStop("SAFE_STOP");
   662→    }
   663→  }
   664→
   665→  private tickStopping(dt: number) {
   666→    const res = stepPhysics(
   667→      this.kin.speed,
   668→      this.kin.position,
   669→      this.kin.lateral,
   670→      this.kin.odometer,
   671→      VEHICLE_SPECS[this.vehicleType],
   672→      { throttle: 0, brake: this.pendingBrake, steer: 0, powerLimit: 1 },
   673→      dt,
   674→      this.pendingLateralTarget,
   675→    );
   676→    this.applyPhysics(res);
   677→    if (this.kin.speed <= STOP_SPEED_EPS) {
   678→      this.kin.speed = 0;
   679→      this.finalizeStop(this.pendingStopResult);
   680→    }
   681→  }
   682→
   683→  private tickEmergency(dt: number) {
   684→    const res = stepPhysics(
   685→      this.kin.speed,
   686→      this.kin.position,
   687→      this.kin.lateral,
   688→      this.kin.odometer,
   689→      VEHICLE_SPECS[this.vehicleType],
   690→      { throttle: 0, brake: 1, steer: 0, powerLimit: 1 },
   691→      dt,
   692→      null,
   693→    );
   694→    this.applyPhysics(res);
   695→    if (this.kin.speed <= STOP_SPEED_EPS) {
   696→      this.kin.speed = 0;
   697→      this.finalizeStop("EMERGENCY_STOP");
   698→    }
   699→  }

   701→  private pendingStopResult: SessionRecord["result"] = "SAFE_STOP";
   702→  private pendingLateralTarget: number | null = null;
   703→  private pendingBrake = 0.55;
   704→
   705→  private beginControlledStop(
   706→    result: SessionRecord["result"],
   707→    lateralTarget: number | null = null,
   708→    brake = 0.55,
   709→  ) {
   710→    if (this.mode === "STOPPING" || this.mode === "STOPPED") return;
   711→    this.pendingStopResult = result;
   712→    this.pendingLateralTarget = lateralTarget;
   713→    this.pendingBrake = brake;
   714→    this.mode = "STOPPING";
   715→    this.hazardsOn = true;
   716→    this.controlLocked = true;
   717→    this.disengageAutopilot("safety intervention");
   718→    if (!this.sessionReason) {
   719→      this.sessionReason =
   720→        this.decision.trigger === null
   721→          ? "Operator requested stop"
   722→          : this.decision.reason;
   723→    }
   724→    this.emit("AI", "SAFETY ENGINE", "Controlled stop engaged — manual control disabled");
   725→  }
   726→
   727→  private enterEmergency(reason: string) {
   728→    this.mode = "EMERGENCY";
   729→    this.controlLocked = true;
   730→    this.hazardsOn = true;
   731→    this.disengageAutopilot("emergency stop");
   732→    this.safetyStatus = "EMERGENCY";
   733→    this.sessionReason = reason;
   734→    this.decision = {
   735→      severity: "EMERGENCY",
   736→      reason,
   737→      recommendedAction: "EMERGENCY_STOP",
   738→      controlMode: "LOCKED",
   739→      evidence: [],
   740→      trigger: "EMERGENCY",
   741→    };
   742→  }
   743→
   744→  private onAutonomyPhase(phase: AutonomyPhase) {
   745→    if (phase === "DECELERATION") {
   746→      this.emit("AI", "SAFETY ENGINE", "Autonomous safety mode active — reducing speed");
   747→    } else if (phase === "ROADSIDE_ALIGNMENT") {
   748→      this.emit("AI", "SAFETY ENGINE", "Roadside alignment — trajectory selected");
   749→    } else if (phase === "SAFE_STOP") {
   750→      this.emit("AI", "SAFETY ENGINE", "Safe stopping zone reached — final braking");
   751→    }
   752→  }
   753→
   754→  private finalizeStop(result: SessionRecord["result"]) {
   755→    this.mode = "STOPPED";
   756→    this.kin.speed = 0;
   757→    this.hazardsOn = true;
   758→    this.controlLocked = true;
   759→    if (result === "EMERGENCY_STOP") {
   760→      this.safetyStatus = "EMERGENCY";
   761→      this.emit("CRITICAL", "VEHICLE ENGINE", "Vehicle stopped — EMERGENCY");
   762→    } else {
   763→      this.safetyStatus = "SAFE";
   764→      this.emit("SAFE", "VEHICLE ENGINE", "Vehicle stopped safely in designated zone");
   765→      playCue("SAFE_STOP");
   766→    }
   767→    if (result === "SAFE_STOP") this.sessionCounts.safeStops += 1;
   768→    this.stoppedByIntervention = true;
   769→    this.emitFinalReport(result);
   770→  }

   772→  /* ------------------------------ vision -------------------------------- */

   774→  private updateDriverAndDrowsiness(dt: number, s: AppSettings) {
   775→    const visionActive = this.vision.active;
   776→    const source: DetectionSource = this.simDrowsiness
   777→      ? "SIMULATION"
   778→      : visionActive
   779→        ? "CAMERA"
   780→        : "NONE";
   781→
   782→    const eyeClosed =
   783→      this.simDrowsiness || (visionActive && s.drowsinessEnabled && this.vision.eyeClosed);
   784→
   785→    // temporal eye-closure accumulator
   786→    if (eyeClosed) {
   787→      this.drowsiness.closureDuration += dt;
   788→      this.drowsiness.eyeClosed = true;
   789→      this.drowsiness.source = source;
   790→      const warn = Math.max(0.6, s.warningDuration);
   791→      const crit = Math.max(warn + 0.4, s.criticalDuration);
   792→      this.drowsiness.level =
   793→        this.drowsiness.closureDuration >= crit
   794→          ? "CRITICAL"
   795→          : this.drowsiness.closureDuration >= warn
   796→            ? "WARNING"
   797→            : this.drowsiness.closureDuration > 0.18
   798→              ? "BLINK"
   799→              : "NORMAL";
   800→      this.drowsiness.score = Math.min(
   801→        100,
   802→        (this.drowsiness.closureDuration / crit) * 100,
   803→      );
   804→      if (this.drowsiness.level === "WARNING" && !this.drowsinessWarnAnnounced) {
   805→        this.drowsinessWarnAnnounced = true;
   806→        this.emit(
   807→          "WARNING",
   808→          source === "CAMERA" ? "AI VISION" : "SIMULATION",
   809→          `Drowsiness warning — eyes closed ${this.drowsiness.closureDuration.toFixed(1)} s`,
   810→        );
   811→        playCue("WARNING");
   812→      }
   813→      if (this.drowsiness.level === "CRITICAL" && !this.criticalAnnounced) {
   814→        this.criticalAnnounced = true;
   815→        this.emit(
   816→          "CRITICAL",
   817→          source === "CAMERA" ? "AI VISION" : "SIMULATION",
   818→          `Critical drowsiness — eyes closed ${this.drowsiness.closureDuration.toFixed(1)} s, driver unresponsive`,
   819→        );
   820→        playCue("CRITICAL");
   821→      }
   822→    } else {
   823→      if (this.drowsiness.closureDuration > 0.18) {
   824→        this.drowsiness.blinkCount += 1;
   825→      }
   826→      const wasWarning =
   827→        this.drowsiness.level === "WARNING" || this.drowsiness.level === "CRITICAL";
   828→      if (wasWarning) {
   829→        this.emit("INFO", "SYSTEM", "Driver alertness restored — drowsiness cleared");
   830→      }
   831→      this.drowsiness.closureDuration = 0;
   832→      this.drowsiness.eyeClosed = false;
   833→      this.drowsiness.level = "NORMAL";
   834→      this.drowsiness.score = Math.max(0, this.drowsiness.score - dt * 60);
   835→      this.drowsinessWarnAnnounced = false;
   836→      this.criticalAnnounced = false;
   837→      if (this.simDrowsiness) {
   838→        // sim toggle on but treated as open (shouldn't happen) — keep closed flag
   839→        this.drowsiness.eyeClosed = true;
   840→      }
   841→    }
   842→    this.drowsiness.source = this.simDrowsiness
   843→      ? "SIMULATION"
   844→      : visionActive
   845→        ? "CAMERA"
   846→        : "NONE";
   847→
   848→    // driver presence
   849→    if (visionActive && s.driverPresenceDetection) {
   850→      this.driver.faceDetected = this.vision.faceDetected;
   851→      this.driver.source = "CAMERA";
   852→      if (!this.vision.faceDetected) {
   853→        if (this.faceAbsentSince === null) this.faceAbsentSince = this.clock;
   854→        if (
   855→          !this.faceAbsentAnnounced &&
   856→          this.clock - this.faceAbsentSince > s.faceAbsenceTimeout
   857→        ) {
   858→          this.faceAbsentAnnounced = true;
   859→          this.emit("WARNING", "AI VISION", "Driver face not detected by camera");
   860→        }
   861→      } else {
   862→        if (this.faceAbsentAnnounced) {
   863→          this.emit("INFO", "AI VISION", "Driver face re-detected");
   864→        }
   865→        this.faceAbsentSince = null;
   866→        this.faceAbsentAnnounced = false;
   867→      }
   868→    } else {
   869→      this.driver.faceDetected = true;
   870→      this.driver.source = "NONE";
   871→      this.faceAbsentSince = null;
   872→      this.faceAbsentAnnounced = false;
   873→    }
   874→    this.driver.present =
   875→      this.driver.source !== "CAMERA" || this.driver.faceDetected;
   876→    const attentionTarget =
   877→      this.drowsiness.level === "CRITICAL"
   878→        ? 12
   879→        : this.drowsiness.level === "WARNING"
   880→          ? 45
   881→          : this.drowsiness.level === "BLINK"
   882→            ? 88
   883→            : !this.driver.present
   884→              ? 55
   885→              : 97;
   886→    this.driver.attention +=
   887→      (attentionTarget - this.driver.attention) * Math.min(1, dt * 3);
   888→  }

   890→  private alcoholEscalated(): boolean {
   891→    if (this.alcoholAt === null) return false;
   892→    return this.clock - this.alcoholAt >= this.settings.interventionDelay;
   893→  }

   895→  /* --------------------------- safety pulse ------------------------------ */
   896→
   897→  private safetyAccum = 0;
   898→
   899→  private safetyPulse(dt: number, s: AppSettings) {
   900→    this.safetyAccum += dt;
   901→    if (this.safetyAccum < 0.1) return;
   902→    this.safetyAccum = 0;
   903→
   904→    const input = {
   905→      vehicleType: this.vehicleType,
   906→      vehicleMode: this.mode,
   907→      helmet: this.helmet,
   908→      alcohol: this.alcohol,
   909→      alcoholEscalated: this.alcoholEscalated(),
   910→      drowsiness: this.drowsiness,
   911→      driver: this.driver,
   912→      accident: this.accident,
   913→      engineOn: this.engineOn,
   914→      moving: this.kin.speed > 0.5,
   915→    };
   916→    const decision = evaluateSafety(input);
   917→    this.decision = decision;
   918→
   919→    // severity transitions → events + audio
   920→    if (decision.severity !== this.prevSeverity) {
   921→      this.onSeverityChange(this.prevSeverity, decision.severity, decision);
   922→      this.prevSeverity = decision.severity;
   923→    }
   924→
   925→    // periodic warning beeps while in warning
   926→    if (
   927→      decision.severity === "WARNING" &&
   928→      this.mode !== "IDLE" &&
   929→      this.clock - this.lastWarnBeep > 2.2
   930→    ) {
   931→      this.lastWarnBeep = this.clock;
   932→      playCue("WARNING");
   933→    }
   934→
   935→    // apply decision actions to the state machine
   936→    // EMERGENCY applies from any active mode (including autonomous)
   937→    if (
   938→      decision.recommendedAction === "EMERGENCY_STOP" &&
   939→      this.mode !== "IDLE" &&
   940→      this.mode !== "SAFETY_CHECK" &&
   941→      this.mode !== "STOPPED"
   942→    ) {
   943→      this.sessionCounts.interventions += 1;
   944→      this.sessionIntervention = "Emergency stop";
   945→      this.enterEmergency("Accident / impact detected");
   946→      return;
   947→    }
   948→
   949→    if (this.mode === "READY" || this.mode === "MANUAL" || this.mode === "WARNING") {
   950→      switch (decision.recommendedAction) {
   951→        case "WARN":
   952→          this.controlLocked = false;
   953→          this.safetyStatus = "WARNING";
   954→          break;
   955→        case "REDUCE_POWER": {
   956→          // bike: motor power reduced → glide to a controlled stop
   957→          this.safetyStatus = "CRITICAL";
   958→          this.sessionCounts.interventions += 1;
   959→          this.sessionIntervention = "Motor power reduction + controlled stop";
   960→          this.bikeMotorLimited = true;
   961→          this.powerLimit = 0.25;
   962→          this.emit(
   963→            "WARNING",
   964→            "SAFETY ENGINE",
   965→            "Motor power reduced — controlled shutdown",
   966→          );
   967→          if (this.kin.speed > 0.2) {
   968→            this.beginControlledStop("SAFE_STOP", null, 0.35);
   969→          } else {
   970→            this.finalizeStop("SAFE_STOP");
   971→          }
   972→          break;
   973→        }
   974→        case "CONTROLLED_STOP":
   975→          this.safetyStatus = "CRITICAL";
   976→          this.sessionCounts.interventions += 1;
   977→          this.sessionIntervention = "Controlled safety stop";
   978→          this.beginControlledStop("SAFE_STOP");
   979→          break;
   980→        case "AUTONOMOUS_STOP":
   981→          this.safetyStatus = "CRITICAL";
   982→          this.beginAutonomousStop(s);
   983→          break;
   984→        case "BLOCK_START":
   985→          // engine running but interlock demands hold (e.g., alcohol while idle)
   986→          this.safetyStatus = "CRITICAL";
   987→          if (this.engineOn) {
   988→            this.beginControlledStop("SAFE_STOP");
   989→          }
   990→          break;
   991→        default:
   992→          this.safetyStatus = "SAFE";
   993→      }
   994→    } else if (
   995→      this.mode === "IDLE" &&
   996→      decision.recommendedAction === "BLOCK_START" &&
   997→      decision.trigger === "HELMET"
   998→    ) {
   999→      // keep the interlock visible on the dashboard
  1000→      this.safetyStatus = "WARNING";
  1001→      this.startBlockedReason = this.startBlockedReason ?? "HELMET NOT DETECTED";
  1002→    }
  1003→  }
  1004→
  1005→  private beginAutonomousStop(s: AppSettings) {
  1006→    if (this.mode === "AUTONOMOUS") return;
  1007→    this.sessionCounts.interventions += 1;
  1008→    this.sessionIntervention = "Autonomous roadside safety stop";
  1009→    this.mode = "AUTONOMOUS";
  1010→    this.controlLocked = true;
  1011→    this.hazardsOn = true;
  1012→    this.disengageAutopilot("autonomous safety takeover");
  1013→    this.firstCriticalAt ??= Date.now();
  1014→    if (!this.takeoverPlanned) {
  1015→      this.takeoverPlanned = true;
  1016→      this.sessionReason = this.decision.reason;
  1017→      planTakeover(
  1018→        this.autonomy,
  1019→        this.kin.speed,
  1020→        this.kin.position,
  1021→        this.kin.lateral,
  1022→        VEHICLE_SPECS[this.vehicleType],
  1023→        s.roadsideTarget,
  1024→      );
  1025→      this.emit(
  1026→        "AI",
  1027→        "SAFETY ENGINE",
  1028→        "AUTONOMOUS TAKEOVER — manual control disabled",
  1029→      );
  1030→      this.emit(
  1031→        "AI",
  1032→        "VEHICLE ENGINE",
  1033→        `Safe stop zone designated at +${Math.round(this.autonomy.stopZone.position - this.kin.position)} m`,
  1034→      );
  1035→      playCue("TAKEOVER");
  1036→    }

  1039→  private onSeverityChange(
  1040→    prev: SafetyStatus,
  1041→    next: SafetyStatus,
  1042→    decision: SafetyDecision,
  1043→  ) {
  1044→    if (next === "CRITICAL" || next === "EMERGENCY") {
  1045→      this.firstCriticalAt ??= Date.now();
  1046→      this.sessionEventTally.critical += 1;
  1047→      if (decision.trigger === "DROWSINESS") {
  1048→        this.sessionCounts.interventions += 0; // counted at action application
  1049→      }
  1050→    } else if (next === "WARNING") {
  1051→      this.sessionEventTally.warning += 1;
  1052→    }
  1053→    this.emit(
  1054→      next === "SAFE"
  1055→        ? "SAFE"
  1056→        : next === "WARNING"
  1057→          ? "WARNING"
  1058→          : "CRITICAL",
  1059→      "SAFETY ENGINE",
  1060→      decision.reason,
  1061→    );
  1062→  }

  1070→    odometer: number;
  1071→    headingDeg: number;

  1074→    this.kin.speed = res.speed;
  1075→    this.kin.position = res.position;
  1076→    this.kin.lateral = res.lateral;
  1077→    this.kin.odometer = res.odometer;

  1083→  emit(type: TimelineEvent["type"], source: TimelineEvent["source"], message: string) {
  1084→    // dedupe identical messages within 1.5 s
  1085→    const key = `${type}:${message}`;
  1086→    const last = this.lastEventAt.get(key) ?? -10;
  1087→    if (this.clock - last < 1.5) return;
  1088→    this.lastEventAt.set(key, this.clock);
  1089→
  1090→    if (type === "INFO") this.sessionEventTally.info += 1;
  1091→    if (type === "WARNING") this.sessionEventTally.warning += 1;
  1092→    if (type === "CRITICAL") this.sessionEventTally.critical += 1;
  1093→    if (type === "AI") this.sessionEventTally.ai += 1;
  1094→
  1095→    const e: TimelineEvent = {
  1096→      id: makeEventId(),
  1097→      ts: Date.now(),
  1098→      type,
  1099→      source,
  1100→      message,
  1101→    };
  1102→    this.listener?.onEvent(e);
  1103→  }

  1105→  private emitFinalReport(result: SessionRecord["result"]) {
  1106→    const durationSec = this.sessionStartedAt
  1107→      ? (Date.now() - this.sessionStartedAt) / 1000
  1108→      : 0;
  1109→    const interventionTimeSec =
  1110→      this.firstCriticalAt && this.sessionStartedAt
  1111→        ? (Date.now() - this.firstCriticalAt) / 1000
  1112→        : null;
  1113→    const record: SessionRecord = {
  1114→      id: makeSessionId(),
  1115→      vehicle: this.vehicleType,
  1116→      startedAt: this.sessionStartedAt ?? Date.now(),
  1117→      durationSec,
  1118→      result,
  1119→      intervention: this.sessionIntervention,
  1120→      interventionTimeSec,
  1121→      reason:
  1122→        this.sessionReason ??
  1123→        (this.decision.trigger
  1124→          ? this.decision.reason
  1125→          : result === "MANUAL_STOP"
  1126→            ? "Operator requested stop"
  1127→            : "Routine stop"),
  1128→      eventCounts: { ...this.sessionEventTally },
  1129→      counts: { ...this.sessionCounts },
  1130→    };
  1131→    this.sessionResult = result;
  1132→    this.listener?.onSessionEnd(record);
  1133→  }
