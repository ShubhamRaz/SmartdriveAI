   400→  tick(rawDt: number) {
   401→    const dt = Math.min(rawDt, 0.05);
   402→    this.clock += dt;
   403→    const s = this.settings;
   404→
   405→    // ---------- vision merge ----------
   406→    this.updateDriverAndDrowsiness(dt, s);
   407→
   408→    // ---------- mode machine ----------
   409→    switch (this.mode) {
   410→      case "SAFETY_CHECK":
   411→        this.tickSafetyCheck(dt);
   412→        break;
   413→      case "READY":
   414→      case "MANUAL":
   415→      case "WARNING":
   416→        this.tickDriving(dt, s);
   417→        break;
   418→      case "AUTONOMOUS":
   419→        this.tickAutonomous(dt);
   420→        break;
   421→      case "STOPPING":
   422→        this.tickStopping(dt);
   423→        break;
   424→      case "EMERGENCY":
   425→        this.tickEmergency(dt);
   426→        break;
   427→      case "STOPPED":
   428→        this.kin.throttle = 0;
   429→        this.kin.brake = 0;
   430→        this.kin.steering = 0;
   431→        break;
   432→      case "IDLE":
   433→        break;
   434→    }
   435→
   436→    // ---------- safety evaluation (10 Hz) ----------
   437→    this.safetyPulse(dt, s);
   438→  }

   442→  private tickSafetyCheck(dt: number) {
   443→    this.safetyCheckTimer += dt;
   444→    if (!this.helmetCheckAnnounced && this.safetyCheckTimer > 0.5) {
   445→      this.helmetCheckAnnounced = true;
   446→      if (VEHICLE_RULES[this.vehicleType].requireHelmetToStart) {
   447→        this.emit(
   448→          this.helmet.state === "HELMET_DETECTED" ? "INFO" : "WARNING",
   449→          "AI VISION",
   450→          this.helmet.state === "HELMET_DETECTED"
   451→            ? "Helmet detected (simulated detector)"
   452→            : "Helmet NOT detected",
   453→        );
   454→      }
   455→      this.emit(
   456→        this.alcohol.detected ? "CRITICAL" : "INFO",
   457→        "AI VISION",
   458→        this.alcohol.detected
   459→          ? `Alcohol sensor reading ${this.alcohol.level} — above threshold`
   460→          : "Alcohol sensor reading 0 — within safe range",
   461→      );
   462→    }
   463→    if (this.safetyCheckTimer >= 1.6) {
   464→      // finalize checks
   465→      const blockedHelmet =
   466→        VEHICLE_RULES[this.vehicleType].requireHelmetToStart &&
   467→        this.helmet.state !== "HELMET_DETECTED";
   468→      const blockedAlcohol = this.alcohol.detected;
   469→      if (blockedHelmet || blockedAlcohol) {
   470→        this.mode = "IDLE";
   471→        this.engineOn = false;
   472→        this.startBlockedReason = blockedHelmet
   473→          ? "HELMET NOT DETECTED"
   474→          : "ALCOHOL DETECTED";
   475→        this.emit(
   476→          "CRITICAL",
   477→          "SAFETY ENGINE",
   478→          `START BLOCKED — ${this.startBlockedReason}`,
   479→        );
   480→        this.safetyStatus = "CRITICAL";
   481→        this.decision = {
   482→          severity: "CRITICAL",
   483→          reason: blockedHelmet
   484→            ? "Helmet interlock: no helmet detected"
   485→            : "Alcohol interlock: sensor above threshold",
   486→          recommendedAction: "BLOCK_START",
   487→          controlMode: "LOCKED",
   488→          evidence: [],
   489→          trigger: blockedHelmet ? "HELMET" : "ALCOHOL",
   490→        };
   491→        this.sessionEventTally.critical += 1;
   492→        this.prevSeverity = "CRITICAL";
   493→        playCue("BLOCKED");
   494→      } else {
   495→        this.mode = "READY";
   496→        this.engineOn = true;
   497→        this.hazardsOn = false;
   498→        this.sessionStartedAt = Date.now();
   499→        this.emit("SAFE", "SAFETY ENGINE", "Safety checks passed");
   500→        this.emit("INFO", "VEHICLE ENGINE", "Engine started — manual control available");
   501→      }
   502→    }
   503→  }
   504→
   505→  private tickDriving(dt: number, s: AppSettings) {
   506→    // manual / warning driving: gather inputs
   507→    let throttle = this.kin.throttle;
   508→    const brake = this.kin.brake;
   509→    const steer = this.kin.steering;
   510→
   511→    if (this.demoAutoThrottle) {
   512→      const cruise = (s.cruiseSpeed[this.vehicleType] / 3.6) * 0.92;
   513→      throttle = this.kin.speed < cruise ? 1 : 0.05;
   514→    }
   515→
   516→    const res = stepPhysics(
   517→      this.kin.speed,
   518→      this.kin.position,
   519→      this.kin.lateral,
   520→      this.kin.odometer,
   521→      VEHICLE_SPECS[this.vehicleType],
   522→      { throttle, brake, steer, powerLimit: this.powerLimit },
   523→      dt,
   524→      null,
   525→    );
   526→    this.applyPhysics(res);
   527→
   528→    // mode semantics
   529→    if (this.decision.severity === "WARNING") {
   530→      this.mode = "WARNING";
   531→    } else {
   532→      this.mode = this.kin.speed > 0.5 ? "MANUAL" : "READY";
   533→    }
   534→
   535→    if (this.manualStopRequested) {
   536→      this.manualStopRequested = false;
   537→      this.beginControlledStop("MANUAL_STOP");
   538→    }
   539→  }
   540→
   541→  private tickAutonomous(dt: number) {
   542→    const { cmd, done, phaseChanged } = autonomyTick(
   543→      this.autonomy,
   544→      this.kin.speed,
   545→      this.kin.lateral,
   546→      this.kin.position,
   547→      VEHICLE_SPECS[this.vehicleType],
   548→      dt,
   549→    );
   550→    if (phaseChanged) this.onAutonomyPhase(phaseChanged);
   551→
   552→    const res = stepPhysics(
   553→      this.kin.speed,
   554→      this.kin.position,
   555→      this.kin.lateral,
   556→      this.kin.odometer,
   557→      VEHICLE_SPECS[this.vehicleType],
   558→      {
   559→        throttle: cmd.throttle,
   560→        brake: cmd.brake,
   561→        steer: 0,
   562→        powerLimit: 1,
   563→      },
   564→      dt,
   565→      cmd.lateralTarget,
   566→    );
   567→    this.applyPhysics(res);
   568→
   569→    if (done) {
   570→      this.finalizeStop("SAFE_STOP");
   571→    }
   572→  }
   573→
   574→  private tickStopping(dt: number) {
   575→    const res = stepPhysics(
   576→      this.kin.speed,
   577→      this.kin.position,
   578→      this.kin.lateral,
   579→      this.kin.odometer,
   580→      VEHICLE_SPECS[this.vehicleType],
   581→      { throttle: 0, brake: this.pendingBrake, steer: 0, powerLimit: 1 },
   582→      dt,
   583→      this.pendingLateralTarget,
   584→    );
   585→    this.applyPhysics(res);
   586→    if (this.kin.speed <= STOP_SPEED_EPS) {
   587→      this.kin.speed = 0;
   588→      this.finalizeStop(this.pendingStopResult);
   589→    }
   590→  }
   591→
   592→  private tickEmergency(dt: number) {
   593→    const res = stepPhysics(
   594→      this.kin.speed,
   595→      this.kin.position,
   596→      this.kin.lateral,
   597→      this.kin.odometer,
   598→      VEHICLE_SPECS[this.vehicleType],
   599→      { throttle: 0, brake: 1, steer: 0, powerLimit: 1 },
   600→      dt,
   601→      null,
   602→    );
   603→    this.applyPhysics(res);
   604→    if (this.kin.speed <= STOP_SPEED_EPS) {
   605→      this.kin.speed = 0;
   606→      this.finalizeStop("EMERGENCY_STOP");
   607→    }
   608→  }
   609→
   610→  private pendingStopResult: SessionRecord["result"] = "SAFE_STOP";
   611→  private pendingLateralTarget: number | null = null;
   612→  private pendingBrake = 0.55;
   613→
   614→  private beginControlledStop(
   615→    result: SessionRecord["result"],
   616→    lateralTarget: number | null = null,
   617→    brake = 0.55,
   618→  ) {
   619→    if (this.mode === "STOPPING" || this.mode === "STOPPED") return;
   620→    this.pendingStopResult = result;
   621→    this.pendingLateralTarget = lateralTarget;
   622→    this.pendingBrake = brake;
   623→    this.mode = "STOPPING";
   624→    this.hazardsOn = true;
   625→    this.controlLocked = true;
   626→    if (!this.sessionReason) {
   627→      this.sessionReason =
   628→        this.decision.trigger === null
   629→          ? "Operator requested stop"
   630→          : this.decision.reason;
   631→    }
   632→    this.emit("AI", "SAFETY ENGINE", "Controlled stop engaged — manual control disabled");
   633→  }
   634→
   635→  private enterEmergency(reason: string) {
   636→    this.mode = "EMERGENCY";
   637→    this.controlLocked = true;
   638→    this.hazardsOn = true;
   639→    this.safetyStatus = "EMERGENCY";
   640→    this.sessionReason = reason;
   641→    this.decision = {
   642→      severity: "EMERGENCY",
   643→      reason,
   644→      recommendedAction: "EMERGENCY_STOP",
   645→      controlMode: "LOCKED",
   646→      evidence: [],
   647→      trigger: "EMERGENCY",
   648→    };
   649→  }

   651→  private onAutonomyPhase(phase: AutonomyPhase) {
   652→    if (phase === "DECELERATION") {
   653→      this.emit("AI", "SAFETY ENGINE", "Autonomous safety mode active — reducing speed");
   654→    } else if (phase === "ROADSIDE_ALIGNMENT") {
   655→      this.emit("AI", "SAFETY ENGINE", "Roadside alignment — trajectory selected");
   656→    } else if (phase === "SAFE_STOP") {
   657→      this.emit("AI", "SAFETY ENGINE", "Safe stopping zone reached — final braking");
   658→    }
   659→  }

   661→  private finalizeStop(result: SessionRecord["result"]) {
   662→    this.mode = "STOPPED";
   663→    this.kin.speed = 0;
   664→    this.hazardsOn = true;
   665→    this.controlLocked = true;
   666→    if (result === "EMERGENCY_STOP") {
   667→      this.safetyStatus = "EMERGENCY";
   668→      this.emit("CRITICAL", "VEHICLE ENGINE", "Vehicle stopped — EMERGENCY");
   669→    } else {
   670→      this.safetyStatus = "SAFE";
   671→      this.emit("SAFE", "VEHICLE ENGINE", "Vehicle stopped safely in designated zone");
   672→      playCue("SAFE_STOP");
   673→    }
   674→    if (result === "SAFE_STOP") this.sessionCounts.safeStops += 1;
   675→    this.stoppedByIntervention = true;
   676→    this.emitFinalReport(result);
   677→  }

   679→  /* ------------------------------ vision -------------------------------- */

   681→  private updateDriverAndDrowsiness(dt: number, s: AppSettings) {
   682→    const visionActive = this.vision.active;
   683→    const source: DetectionSource = this.simDrowsiness
   684→      ? "SIMULATION"
   685→      : visionActive
   686→        ? "CAMERA"
   687→        : "NONE";
   688→
   689→    const eyeClosed =
   690→      this.simDrowsiness || (visionActive && s.drowsinessEnabled && this.vision.eyeClosed);
   691→
   692→    // temporal eye-closure accumulator
   693→    if (eyeClosed) {
   694→      this.drowsiness.closureDuration += dt;
   695→      this.drowsiness.eyeClosed = true;
   696→      this.drowsiness.source = source;
   697→      const warn = Math.max(0.6, s.warningDuration);
   698→      const crit = Math.max(warn + 0.4, s.criticalDuration);
   699→      this.drowsiness.level =
   700→        this.drowsiness.closureDuration >= crit
   701→          ? "CRITICAL"
   702→          : this.drowsiness.closureDuration >= warn
   703→            ? "WARNING"
   704→            : this.drowsiness.closureDuration > 0.18
   705→              ? "BLINK"
   706→              : "NORMAL";
   707→      this.drowsiness.score = Math.min(
   708→        100,
   709→        (this.drowsiness.closureDuration / crit) * 100,
   710→      );
   711→      if (this.drowsiness.level === "WARNING" && !this.drowsinessWarnAnnounced) {
   712→        this.drowsinessWarnAnnounced = true;
   713→        this.emit(
   714→          "WARNING",
   715→          source === "CAMERA" ? "AI VISION" : "SIMULATION",
   716→          `Drowsiness warning — eyes closed ${this.drowsiness.closureDuration.toFixed(1)} s`,
   717→        );
   718→        playCue("WARNING");
   719→      }
   720→      if (this.drowsiness.level === "CRITICAL" && !this.criticalAnnounced) {
   721→        this.criticalAnnounced = true;
   722→        this.emit(
   723→          "CRITICAL",
   724→          source === "CAMERA" ? "AI VISION" : "SIMULATION",
   725→          `Critical drowsiness — eyes closed ${this.drowsiness.closureDuration.toFixed(1)} s, driver unresponsive`,
   726→        );
   727→        playCue("CRITICAL");
   728→      }
   729→    } else {
   730→      if (this.drowsiness.closureDuration > 0.18) {
   731→        this.drowsiness.blinkCount += 1;
   732→      }
   733→      const wasWarning =
   734→        this.drowsiness.level === "WARNING" || this.drowsiness.level === "CRITICAL";
   735→      if (wasWarning) {
   736→        this.emit("INFO", "SYSTEM", "Driver alertness restored — drowsiness cleared");
   737→      }
   738→      this.drowsiness.closureDuration = 0;
   739→      this.drowsiness.eyeClosed = false;
   740→      this.drowsiness.level = "NORMAL";
   741→      this.drowsiness.score = Math.max(0, this.drowsiness.score - dt * 60);
   742→      this.drowsinessWarnAnnounced = false;
   743→      this.criticalAnnounced = false;
   744→      if (this.simDrowsiness) {
   745→        // sim toggle on but treated as open (shouldn't happen) — keep closed flag
   746→        this.drowsiness.eyeClosed = true;
   747→      }
   748→    }
   749→    this.drowsiness.source = this.simDrowsiness
   750→      ? "SIMULATION"
   751→      : visionActive
   752→        ? "CAMERA"
   753→        : "NONE";
   754→
   755→    // driver presence
   756→    if (visionActive && s.driverPresenceDetection) {
   757→      this.driver.faceDetected = this.vision.faceDetected;
   758→      this.driver.source = "CAMERA";
   759→      if (!this.vision.faceDetected) {
   760→        if (this.faceAbsentSince === null) this.faceAbsentSince = this.clock;
   761→        if (
   762→          !this.faceAbsentAnnounced &&
   763→          this.clock - this.faceAbsentSince > s.faceAbsenceTimeout
   764→        ) {
   765→          this.faceAbsentAnnounced = true;
   766→          this.emit("WARNING", "AI VISION", "Driver face not detected by camera");
   767→        }
   768→      } else {
   769→        if (this.faceAbsentAnnounced) {
   770→          this.emit("INFO", "AI VISION", "Driver face re-detected");
   771→        }
   772→        this.faceAbsentSince = null;
   773→        this.faceAbsentAnnounced = false;
   774→      }
   775→    } else {
   776→      this.driver.faceDetected = true;
   777→      this.driver.source = "NONE";
   778→      this.faceAbsentSince = null;
   779→      this.faceAbsentAnnounced = false;
   780→    }
   781→    this.driver.present =
   782→      this.driver.source !== "CAMERA" || this.driver.faceDetected;
   783→    const attentionTarget =
   784→      this.drowsiness.level === "CRITICAL"
   785→        ? 12
   786→        : this.drowsiness.level === "WARNING"
   787→          ? 45
   788→          : this.drowsiness.level === "BLINK"
   789→            ? 88
   790→            : !this.driver.present
   791→              ? 55
   792→              : 97;
   793→    this.driver.attention +=
   794→      (attentionTarget - this.driver.attention) * Math.min(1, dt * 3);
   795→  }

   797→  private alcoholEscalated(): boolean {
   798→    if (this.alcoholAt === null) return false;
   799→    return this.clock - this.alcoholAt >= this.settings.interventionDelay;
   800→  }

   802→  /* --------------------------- safety pulse ------------------------------ */
   803→
   804→  private safetyAccum = 0;
   805→
   806→  private safetyPulse(dt: number, s: AppSettings) {
   807→    this.safetyAccum += dt;
   808→    if (this.safetyAccum < 0.1) return;
   809→    this.safetyAccum = 0;
   810→
   811→    const input = {
   812→      vehicleType: this.vehicleType,
   813→      vehicleMode: this.mode,
   814→      helmet: this.helmet,
   815→      alcohol: this.alcohol,
   816→      alcoholEscalated: this.alcoholEscalated(),
   817→      drowsiness: this.drowsiness,
   818→      driver: this.driver,
   819→      accident: this.accident,
   820→      engineOn: this.engineOn,
   821→      moving: this.kin.speed > 0.5,
   822→    };
   823→    const decision = evaluateSafety(input);
   824→    this.decision = decision;
   825→
   826→    // severity transitions → events + audio
   827→    if (decision.severity !== this.prevSeverity) {
   828→      this.onSeverityChange(this.prevSeverity, decision.severity, decision);
   829→      this.prevSeverity = decision.severity;
   830→    }
   831→
   832→    // periodic warning beeps while in warning
   833→    if (
   834→      decision.severity === "WARNING" &&
   835→      this.mode !== "IDLE" &&
   836→      this.clock - this.lastWarnBeep > 2.2
   837→    ) {
   838→      this.lastWarnBeep = this.clock;
   839→      playCue("WARNING");
   840→    }
   841→
   842→    // apply decision actions to the state machine
   843→    // EMERGENCY applies from any active mode (including autonomous)
   844→    if (
   845→      decision.recommendedAction === "EMERGENCY_STOP" &&
   846→      this.mode !== "IDLE" &&
   847→      this.mode !== "SAFETY_CHECK" &&
   848→      this.mode !== "STOPPED"
   849→    ) {
   850→      this.sessionCounts.interventions += 1;
   851→      this.sessionIntervention = "Emergency stop";
   852→      this.enterEmergency("Accident / impact detected");
   853→      return;
   854→    }
   855→
   856→    if (this.mode === "READY" || this.mode === "MANUAL" || this.mode === "WARNING") {
   857→      switch (decision.recommendedAction) {
   858→        case "WARN":
   859→          this.controlLocked = false;
   860→          this.safetyStatus = "WARNING";
   861→          break;
   862→        case "REDUCE_POWER": {
   863→          // bike: motor power reduced → glide to a controlled stop
   864→          this.safetyStatus = "CRITICAL";
   865→          this.sessionCounts.interventions += 1;
   866→          this.sessionIntervention = "Motor power reduction + controlled stop";
   867→          this.bikeMotorLimited = true;
   868→          this.powerLimit = 0.25;
   869→          this.emit(
   870→            "WARNING",
   871→            "SAFETY ENGINE",
   872→            "Motor power reduced — controlled shutdown",
   873→          );
   874→          if (this.kin.speed > 0.2) {
   875→            this.beginControlledStop("SAFE_STOP", null, 0.35);
   876→          } else {
   877→            this.finalizeStop("SAFE_STOP");
   878→          }
   879→          break;
   880→        }
   881→        case "CONTROLLED_STOP":
   882→          this.safetyStatus = "CRITICAL";
   883→          this.sessionCounts.interventions += 1;
   884→          this.sessionIntervention = "Controlled safety stop";
   885→          this.beginControlledStop("SAFE_STOP");
   886→          break;
   887→        case "AUTONOMOUS_STOP":
   888→          this.safetyStatus = "CRITICAL";
   889→          this.beginAutonomousStop(s);
   890→          break;
   891→        case "BLOCK_START":
   892→          // engine running but interlock demands hold (e.g., alcohol while idle)
   893→          this.safetyStatus = "CRITICAL";
   894→          if (this.engineOn) {
   895→            this.beginControlledStop("SAFE_STOP");
   896→          }
   897→          break;
   898→        default:
   899→          this.safetyStatus = this.mode === "STOPPED" ? this.safetyStatus : "SAFE";
   900→      }
   901→    } else if (
   902→      this.mode === "IDLE" &&
   903→      decision.recommendedAction === "BLOCK_START" &&
   904→      decision.trigger === "HELMET"
   905→    ) {
   906→      // keep the interlock visible on the dashboard
   907→      this.safetyStatus = "WARNING";
   908→      this.startBlockedReason = this.startBlockedReason ?? "HELMET NOT DETECTED";
   909→    }
   910→  }
   911→
   912→  private beginAutonomousStop(s: AppSettings) {
   913→    if (this.mode === "AUTONOMOUS") return;
   914→    this.sessionCounts.interventions += 1;
   915→    this.sessionIntervention = "Autonomous roadside safety stop";
   916→    this.mode = "AUTONOMOUS";
   917→    this.controlLocked = true;
   918→    this.hazardsOn = true;
   919→    this.firstCriticalAt ??= Date.now();
   920→    if (!this.takeoverPlanned) {
   921→      this.takeoverPlanned = true;
   922→      this.sessionReason = this.decision.reason;
   923→      planTakeover(
   924→        this.autonomy,
   925→        this.kin.speed,
   926→        this.kin.position,
   927→        this.kin.lateral,
   928→        VEHICLE_SPECS[this.vehicleType],
   929→        s.roadsideTarget,
   930→      );
   931→      this.emit(
   932→        "AI",
   933→        "SAFETY ENGINE",
   934→        "AUTONOMOUS TAKEOVER — manual control disabled",
   935→      );
   936→      this.emit(
   937→        "AI",
   938→        "VEHICLE ENGINE",
   939→        `Safe stop zone designated at +${Math.round(this.autonomy.stopZone.position - this.kin.position)} m`,
   940→      );
   941→      playCue("TAKEOVER");
   942→    }
   943→  }

   945→  private onSeverityChange(
   946→    prev: SafetyStatus,
   947→    next: SafetyStatus,
   948→    decision: SafetyDecision,
   949→  ) {
   950→    if (next === "CRITICAL" || next === "EMERGENCY") {
   951→      this.firstCriticalAt ??= Date.now();
   952→      this.sessionEventTally.critical += 1;
   953→      if (decision.trigger === "DROWSINESS") {
   954→        this.sessionCounts.interventions += 0; // counted at action application
   955→      }
   956→    } else if (next === "WARNING") {
   957→      this.sessionEventTally.warning += 1;
   958→    }
   959→    this.emit(
   960→      next === "SAFE"
   961→        ? "SAFE"
   962→        : next === "WARNING"
   963→          ? "WARNING"
   964→          : "CRITICAL",
   965→      "SAFETY ENGINE",
   966→      decision.reason,
   967→    );
   968→  }

   988→
   989→  emit(type: TimelineEvent["type"], source: TimelineEvent["source"], message: string) {
   990→    // dedupe identical messages within 1.5 s
   991→    const key = `${type}:${message}`;
   992→    const last = this.lastEventAt.get(key) ?? -10;
   993→    if (this.clock - last < 1.5) return;
   994→    this.lastEventAt.set(key, this.clock);
   995→
   996→    if (type === "INFO") this.sessionEventTally.info += 1;
   997→    if (type === "WARNING") this.sessionEventTally.warning += 1;
   998→    if (type === "CRITICAL") this.sessionEventTally.critical += 1;
   999→    if (type === "AI") this.sessionEventTally.ai += 1;
  1000→
  1001→    const e: TimelineEvent = {
  1002→      id: makeEventId(),
  1003→      ts: Date.now(),
  1004→      type,
  1005→      source,
  1006→      message,
  1007→    };
  1008→    this.listener?.onEvent(e);
  1009→  }

  1011→  private emitFinalReport(result: SessionRecord["result"]) {
  1012→    const durationSec = this.sessionStartedAt
  1013→      ? (Date.now() - this.sessionStartedAt) / 1000
  1014→      : 0;
  1015→    const interventionTimeSec =
  1016→      this.firstCriticalAt && this.sessionStartedAt
  1017→        ? (Date.now() - this.firstCriticalAt) / 1000
  1018→        : null;
  1019→    const record: SessionRecord = {
  1020→      id: makeSessionId(),
  1021→      vehicle: this.vehicleType,
  1022→      startedAt: this.sessionStartedAt ?? Date.now(),
  1023→      durationSec,
  1024→      result,
  1025→      intervention: this.sessionIntervention,
  1026→      interventionTimeSec,
  1027→      reason:
  1028→        this.sessionReason ??
  1029→        (this.decision.trigger
  1030→          ? this.decision.reason
  1031→          : result === "MANUAL_STOP"
  1032→            ? "Operator requested stop"
  1033→            : "Routine stop"),
  1034→      eventCounts: { ...this.sessionEventTally },
  1035→      counts: { ...this.sessionCounts },
  1036→    };
  1037→    this.sessionResult = result;
  1038→    this.listener?.onSessionEnd(record);
  1039→  }
