   441→   441→
   442→   442→  tick(rawDt: number) {
   443→   443→    const dt = Math.min(rawDt, 0.05);
   444→   444→    this.clock += dt;
   445→   445→    const s = this.settings;
   446→   446→
   447→   447→    // ---------- vision merge ----------
   448→   448→    this.updateDriverAndDrowsiness(dt, s);
   449→   449→
   450→   450→    // ---------- mode machine ----------
   451→   451→    switch (this.mode) {
   452→   452→      case "SAFETY_CHECK":
   453→   453→        this.tickSafetyCheck(dt);
   454→   454→        break;
   455→   455→      case "READY":
   456→   456→      case "MANUAL":
   457→   457→      case "WARNING":
   458→   458→        if (this.autopilot) {
   459→   459→          this.tickAutopilot(dt);
   460→   460→        } else {
   461→   461→          this.tickDriving(dt, s);
   462→   462→        }
   463→   463→        break;
   464→   464→      case "AUTONOMOUS":
   465→   465→        this.tickAutonomous(dt);
   466→   466→        break;
   467→   467→      case "STOPPING":
   468→   468→        this.tickStopping(dt);
   469→   469→        break;
   470→   470→      case "EMERGENCY":
   471→   471→        this.tickEmergency(dt);
   472→   472→        break;
   473→   473→      case "STOPPED":
   474→   474→        this.kin.throttle = 0;
   475→   475→        this.kin.brake = 0;
   476→   476→        this.kin.steering = 0;
   477→   477→        break;
   478→   478→      case "IDLE":
   479→   479→        break;
   480→   480→    }
   481→   481→
   482→   482→    // ---------- safety evaluation (10 Hz) ----------
   483→   483→    this.safetyPulse(dt, s);
   484→   484→  }

   487→   487→
   488→   488→  private tickSafetyCheck(dt: number) {
   489→   489→    this.safetyCheckTimer += dt;
   490→   490→    if (!this.helmetCheckAnnounced && this.safetyCheckTimer > 0.5) {
   491→   491→      this.helmetCheckAnnounced = true;
   492→   492→      if (VEHICLE_RULES[this.vehicleType].requireHelmetToStart) {
   493→   493→        this.emit(
   494→   494→          this.helmet.state === "HELMET_DETECTED" ? "INFO" : "WARNING",
   495→   495→          "AI VISION",
   496→   496→          this.helmet.state === "HELMET_DETECTED"
   497→   497→            ? "Helmet detected (simulated detector)"
   498→   498→            : "Helmet NOT detected",
   499→   499→        );
   500→   500→      }
   501→   501→      this.emit(
   502→   502→        this.alcohol.detected ? "CRITICAL" : "INFO",
   503→   503→        "AI VISION",
   504→   504→        this.alcohol.detected
   505→   505→          ? `Alcohol sensor reading ${this.alcohol.level} — above threshold`
   506→   506→          : "Alcohol sensor reading 0 — within safe range",
   507→   507→      );
   508→   508→    }
   509→   509→    if (this.safetyCheckTimer >= 1.6) {
   510→   510→      // finalize checks
   511→   511→      const blockedHelmet =
   512→   512→        VEHICLE_RULES[this.vehicleType].requireHelmetToStart &&
   513→   513→        this.helmet.state !== "HELMET_DETECTED";
   514→   514→      const blockedAlcohol = this.alcohol.detected;
   515→   515→      if (blockedHelmet || blockedAlcohol) {
   516→   516→        this.mode = "IDLE";
   517→   517→        this.engineOn = false;
   518→   518→        this.startBlockedReason = blockedHelmet
   519→   519→          ? "HELMET NOT DETECTED"
   520→   520→          : "ALCOHOL DETECTED";
   521→   521→        this.emit(
   522→   522→          "CRITICAL",
   523→   523→          "SAFETY ENGINE",
   524→   524→          `START BLOCKED — ${this.startBlockedReason}`,
   525→   525→        );
   526→   526→        this.safetyStatus = "CRITICAL";
   527→   527→        this.decision = {
   528→   528→          severity: "CRITICAL",
   529→   529→          reason: blockedHelmet
   530→   530→            ? "Helmet interlock: no helmet detected"
   531→   531→            : "Alcohol interlock: sensor above threshold",
   532→   532→          recommendedAction: "BLOCK_START",
   533→   533→          controlMode: "LOCKED",
   534→   534→          evidence: [],
   535→   535→          trigger: blockedHelmet ? "HELMET" : "ALCOHOL",
   536→   536→        };
   537→   537→        this.sessionEventTally.critical += 1;
   538→   538→        this.prevSeverity = "CRITICAL";
   539→   539→        playCue("BLOCKED");
   540→   540→      } else {
   541→   541→        this.mode = "READY";
   542→   542→        this.engineOn = true;
   543→   543→        this.hazardsOn = false;
   544→   544→        this.sessionStartedAt = Date.now();
   545→   545→        this.emit("SAFE", "SAFETY ENGINE", "Safety checks passed");
   546→   546→        this.emit("INFO", "VEHICLE ENGINE", "Engine started — manual control available");
   547→   547→      }
   548→   548→    }
   549→   549→  }

   551→   551→  private tickDriving(dt: number, s: AppSettings) {
   552→   552→    // manual / warning driving: gather inputs (keyboard channel + on-screen pad)
   553→   553→    let throttle = this.kin.throttle;
   554→   554→    let brake = this.kin.brake;
   555→   555→    let steer = this.kin.steering;
   556→   556→
   557→   557→    if (this.demoAutoThrottle) {
   558→   558→      const cruise = (s.cruiseSpeed[this.vehicleType] / 3.6) * 0.92;
   559→   559→      throttle = this.kin.speed < cruise ? 1 : 0.05;
   560→   560→    } else {
   561→   561→      // merge on-screen pedal input (strongest input wins)
   562→   562→      throttle = Math.max(throttle, this.pad.throttle);
   563→   563→      brake = Math.max(brake, this.pad.brake);
   564→   564→      steer = Math.max(-1, Math.min(1, steer + this.pad.steer));
   565→   565→    }
   566→   566→
   567→   567→    const res = stepPhysics(
   568→   568→      this.kin.speed,
   569→   569→      this.kin.position,
   570→   570→      this.kin.lateral,
   571→   571→      this.kin.odometer,
   572→   572→      VEHICLE_SPECS[this.vehicleType],
   573→   573→      { throttle, brake, steer, powerLimit: this.powerLimit },
   574→   574→      dt,
   575→   575→      null,
   576→   576→    );
   577→   577→    this.applyPhysics(res);
   578→   578→
   579→   579→    // mode semantics
   580→   580→    if (this.decision.severity === "WARNING") {
   581→   581→      this.mode = "WARNING";
   582→   582→    } else {
   583→   583→      this.mode = this.kin.speed > 0.5 ? "MANUAL" : "READY";
   584→   584→    }
   585→   585→
   586→   586→    if (this.manualStopRequested) {
   587→   587→      this.manualStopRequested = false;
   588→   588→      this.beginControlledStop("MANUAL_STOP");
   589→   589→    }
   590→   590→  }
   591→   591→
   592→   592→  /** autopilot chauffeur: cruise control + lane keeping toward lane center */
   593→   593→  private tickAutopilot(dt: number) {
   594→   594→    // driver brake input disengages autopilot (same convention as real ADAS)
   595→   595→    const brakeInput = Math.max(this.kin.brake, this.pad.brake);
   596→   596→    if (brakeInput > 0.2) {
   597→   597→      this.disengageAutopilot("driver brake input");
   598→   598→      this.tickDriving(dt, this.settings);
   599→   599→      return;
   600→   600→    }
   601→   601→
   602→   602→    const cruise = this.settings.cruiseSpeed[this.vehicleType] / 3.6;
   603→   603→    const err = cruise - this.kin.speed;
   604→   604→    const throttle = err > 0.3 ? Math.min(1, 0.25 + err * 0.6) : err < -0.6 ? 0 : 0.08;
   605→   605→    const res = stepPhysics(
   606→   606→      this.kin.speed,
   607→   607→      this.kin.position,
   608→   608→      this.kin.lateral,
   609→   609→      this.kin.odometer,
   610→   610→      VEHICLE_SPECS[this.vehicleType],
   611→   611→      { throttle, brake: 0, steer: 0, powerLimit: this.powerLimit },
   612→   612→      dt,
   613→   613→      0, // lane keeping: steer back to lane center
   614→   614→    );
   615→   615→    this.applyPhysics(res);
   616→   616→
   617→   617→    // mode semantics
   618→   618→    this.mode =
   619→   619→      this.decision.severity === "WARNING"
   620→   620→        ? "WARNING"
   621→   621→        : this.kin.speed > 0.5
   622→   622→          ? "MANUAL"
   623→   623→          : "READY";
   624→   624→
   625→   625→    if (this.manualStopRequested) {
   626→   626→      this.manualStopRequested = false;
   627→   627→      this.disengageAutopilot("operator stop request");
   628→   628→      this.beginControlledStop("MANUAL_STOP");
   629→   629→    }
   630→   630→  }

   639→   639→      dt,
   640→   640→    );

   642→   642→
   643→   643→    const res = stepPhysics(
   644→   644→      this.kin.speed,
   645→   645→      this.kin.position,
   646→   646→      this.kin.lateral,
   647→   647→      this.kin.odometer,

   649→   649→      {
   650→   650→        throttle: cmd.throttle,
   651→   651→        brake: cmd.brake,

   654→   654→      },
   655→   655→      dt,
   656→   656→      cmd.lateralTarget,

   774→   774→  private updateDriverAndDrowsiness(dt: number, s: AppSettings) {
   775→   775→    const visionActive = this.vision.active;
   776→   776→    const source: DetectionSource = this.simDrowsiness
   777→   777→      ? "SIMULATION"
   778→   778→      : visionActive
   779→   779→        ? "CAMERA"
   780→   780→        : "NONE";
   781→   781→
   782→   782→    const eyeClosed =
   783→   783→      this.simDrowsiness || (visionActive && s.drowsinessEnabled && this.vision.eyeClosed);
   784→   784→
   785→   785→    // temporal eye-closure accumulator
   786→   786→    if (eyeClosed) {
   787→   787→      this.drowsiness.closureDuration += dt;
   788→   788→      this.drowsiness.eyeClosed = true;
   789→   789→      this.drowsiness.source = source;
   790→   790→      const warn = Math.max(0.6, s.warningDuration);
   791→   791→      const crit = Math.max(warn + 0.4, s.criticalDuration);
   792→   792→      this.drowsiness.level =
   793→   793→        this.drowsiness.closureDuration >= crit
   794→   794→          ? "CRITICAL"
   795→   795→          : this.drowsiness.closureDuration >= warn
   796→   796→            ? "WARNING"
   797→   797→            : this.drowsiness.closureDuration > 0.18
   798→   798→              ? "BLINK"
   799→   799→              : "NORMAL";
   800→   800→      this.drowsiness.score = Math.min(
   801→   801→        100,
   802→   802→        (this.drowsiness.closureDuration / crit) * 100,
   803→   803→      );
   804→   804→      if (this.drowsiness.level === "WARNING" && !this.drowsinessWarnAnnounced) {
   805→   805→        this.drowsinessWarnAnnounced = true;
   806→   806→        this.emit(
   807→   807→          "WARNING",
   808→   808→          source === "CAMERA" ? "AI VISION" : "SIMULATION",
   809→   809→          `Drowsiness warning — eyes closed ${this.drowsiness.closureDuration.toFixed(1)} s`,
   810→   810→        );
   811→   811→        playCue("WARNING");
   812→   812→      }
   813→   813→      if (this.drowsiness.level === "CRITICAL" && !this.criticalAnnounced) {
   814→   814→        this.criticalAnnounced = true;
   815→   815→        this.emit(
   816→   816→          "CRITICAL",
   817→   817→          source === "CAMERA" ? "AI VISION" : "SIMULATION",
   818→   818→          `Critical drowsiness — eyes closed ${this.drowsiness.closureDuration.toFixed(1)} s, driver unresponsive`,
   819→   819→        );
   820→   820→        playCue("CRITICAL");
   821→   821→      }
   822→   822→    } else {
   823→   823→      if (this.drowsiness.closureDuration > 0.18) {
   824→   824→        this.drowsiness.blinkCount += 1;
   825→   825→      }
   826→   826→      const wasWarning =
   827→   827→        this.drowsiness.level === "WARNING" || this.drowsiness.level === "CRITICAL";
   828→   828→      if (wasWarning) {
   829→   829→        this.emit("INFO", "SYSTEM", "Driver alertness restored — drowsiness cleared");
   830→   830→      }
   831→   831→      this.drowsiness.closureDuration = 0;
   832→   832→      this.drowsiness.eyeClosed = false;
   833→   833→      this.drowsiness.level = "NORMAL";
   834→   834→      this.drowsiness.score = Math.max(0, this.drowsiness.score - dt * 60);
   835→   835→      this.drowsinessWarnAnnounced = false;
   836→   836→      this.criticalAnnounced = false;
   837→   837→      if (this.simDrowsiness) {
   838→   838→        // sim toggle on but treated as open (shouldn't happen) — keep closed flag
   839→   839→        this.drowsiness.eyeClosed = true;
   840→   840→      }
   841→   841→    }
   842→   842→    this.drowsiness.source = this.simDrowsiness
   843→   843→      ? "SIMULATION"
   844→   844→      : visionActive
   845→   845→        ? "CAMERA"
   846→   846→        : "NONE";
   847→   847→
   848→   848→    // driver presence
   849→   849→    if (visionActive && s.driverPresenceDetection) {
   850→   850→      this.driver.faceDetected = this.vision.faceDetected;
   851→   851→      this.driver.source = "CAMERA";
   852→   852→      if (!this.vision.faceDetected) {
   853→   853→        if (this.faceAbsentSince === null) this.faceAbsentSince = this.clock;
   854→   854→        if (
   855→   855→          !this.faceAbsentAnnounced &&
   856→   856→          this.clock - this.faceAbsentSince > s.faceAbsenceTimeout
   857→   857→        ) {
   858→   858→          this.faceAbsentAnnounced = true;
   859→   859→          this.emit("WARNING", "AI VISION", "Driver face not detected by camera");
   860→   860→        }
   861→   861→      } else {
   862→   862→        if (this.faceAbsentAnnounced) {
   863→   863→          this.emit("INFO", "AI VISION", "Driver face re-detected");
   864→   864→        }
   865→   865→        this.faceAbsentSince = null;
   866→   866→        this.faceAbsentAnnounced = false;
   867→   867→      }
   868→   868→    } else {
   869→   869→      this.driver.faceDetected = true;
   870→   870→      this.driver.source = "NONE";
   871→   871→      this.faceAbsentSince = null;
   872→   872→      this.faceAbsentAnnounced = false;
   873→   873→    }
   874→   874→    this.driver.present =
   875→   875→      this.driver.source !== "CAMERA" || this.driver.faceDetected;
   876→   876→    const attentionTarget =
   877→   877→      this.drowsiness.level === "CRITICAL"
   878→   878→        ? 12
   879→   879→        : this.drowsiness.level === "WARNING"
   880→   880→          ? 45
   881→   881→          : this.drowsiness.level === "BLINK"
   882→   882→            ? 88
   883→   883→            : !this.driver.present
   884→   884→              ? 55
   885→   885→              : 97;
   886→   886→    this.driver.attention +=
   887→   887→      (attentionTarget - this.driver.attention) * Math.min(1, dt * 3);
   888→   888→  }

   895→   895→  /* --------------------------- safety pulse ------------------------------ */
   896→   896→
   897→   897→  private safetyAccum = 0;
   898→   898→
   899→   899→  private safetyPulse(dt: number, s: AppSettings) {
   900→   900→    this.safetyAccum += dt;
   901→   901→    if (this.safetyAccum < 0.1) return;
   902→   902→    this.safetyAccum = 0;
   903→   903→
   904→   904→    const input = {
   905→   905→      vehicleType: this.vehicleType,
   906→   906→      vehicleMode: this.mode,
   907→   907→      helmet: this.helmet,
   908→   908→      alcohol: this.alcohol,
   909→   909→      alcoholEscalated: this.alcoholEscalated(),
   910→   910→      drowsiness: this.drowsiness,
   911→   911→      driver: this.driver,
   912→   912→      accident: this.accident,
   913→   913→      engineOn: this.engineOn,
   914→   914→      moving: this.kin.speed > 0.5,
   915→   915→    };
   916→   916→    const decision = evaluateSafety(input);
   917→   917→    this.decision = decision;
   918→   918→
   919→   919→    // severity transitions → events + audio
   920→   920→    if (decision.severity !== this.prevSeverity) {
   921→   921→      this.onSeverityChange(this.prevSeverity, decision.severity, decision);
   922→   922→      this.prevSeverity = decision.severity;
   923→   923→    }
   924→   924→
   925→   925→    // periodic warning beeps while in warning
   926→   926→    if (
   927→   927→      decision.severity === "WARNING" &&
   928→   928→      this.mode !== "IDLE" &&
   929→   929→      this.clock - this.lastWarnBeep > 2.2
   930→   930→    ) {
   931→   931→      this.lastWarnBeep = this.clock;
   932→   932→      playCue("WARNING");
   933→   933→    }
   934→   934→
   935→   935→    // apply decision actions to the state machine
   936→   936→    // EMERGENCY applies from any active mode (including autonomous)
   937→   937→    if (
   938→   938→      decision.recommendedAction === "EMERGENCY_STOP" &&
   939→   939→      this.mode !== "IDLE" &&
   940→   940→      this.mode !== "SAFETY_CHECK" &&
   941→   941→      this.mode !== "STOPPED"
   942→   942→    ) {
   943→   943→      this.sessionCounts.interventions += 1;
   944→   944→      this.sessionIntervention = "Emergency stop";
   945→   945→      this.enterEmergency("Accident / impact detected");
   946→   946→      return;
   947→   947→    }
   948→   948→
   949→   949→    if (this.mode === "READY" || this.mode === "MANUAL" || this.mode === "WARNING") {
   950→   950→      switch (decision.recommendedAction) {
   951→   951→        case "WARN":
   952→   952→          this.controlLocked = false;
   953→   953→          this.safetyStatus = "WARNING";
   954→   954→          break;
   955→   955→        case "REDUCE_POWER": {
   956→   956→          // bike: motor power reduced → glide to a controlled stop
   957→   957→          this.safetyStatus = "CRITICAL";
   958→   958→          this.sessionCounts.interventions += 1;
   959→   959→          this.sessionIntervention = "Motor power reduction + controlled stop";
   960→   960→          this.bikeMotorLimited = true;
   961→   961→          this.powerLimit = 0.25;
   962→   962→          this.emit(
   963→   963→            "WARNING",
   964→   964→            "SAFETY ENGINE",
   965→   965→            "Motor power reduced — controlled shutdown",
   966→   966→          );
   967→   967→          if (this.kin.speed > 0.2) {
   968→   968→            this.beginControlledStop("SAFE_STOP", null, 0.35);
   969→   969→          } else {
   970→   970→            this.finalizeStop("SAFE_STOP");
   971→   971→          }
   972→   972→          break;
   973→   973→        }
   974→   974→        case "CONTROLLED_STOP":
   975→   975→          this.safetyStatus = "CRITICAL";
   976→   976→          this.sessionCounts.interventions += 1;
   977→   977→          this.sessionIntervention = "Controlled safety stop";
   978→   978→          this.beginControlledStop("SAFE_STOP");
   979→   979→          break;
   980→   980→        case "AUTONOMOUS_STOP":
   981→   981→          this.safetyStatus = "CRITICAL";
   982→   982→          this.beginAutonomousStop(s);
   983→   983→          break;
   984→   984→        case "BLOCK_START":
   985→   985→          // engine running but interlock demands hold (e.g., alcohol while idle)
   986→   986→          this.safetyStatus = "CRITICAL";
   987→   987→          if (this.engineOn) {
   988→   988→            this.beginControlledStop("SAFE_STOP");
   989→   989→          }
   990→   990→          break;
   991→   991→        default:
   992→   992→          this.safetyStatus = "SAFE";
   993→   993→      }
   994→   994→    } else if (
   995→   995→      this.mode === "IDLE" &&
   996→   996→      decision.recommendedAction === "BLOCK_START" &&
   997→   997→      decision.trigger === "HELMET"
   998→   998→    ) {
   999→   999→      // keep the interlock visible on the dashboard
  1000→  1000→      this.safetyStatus = "WARNING";
  1001→  1001→      this.startBlockedReason = this.startBlockedReason ?? "HELMET NOT DETECTED";
  1002→  1002→    }
  1003→  1003→  }

  1005→  1005→  private beginAutonomousStop(s: AppSettings) {
  1006→  1006→    if (this.mode === "AUTONOMOUS") return;
  1007→  1007→    this.sessionCounts.interventions += 1;
  1008→  1008→    this.sessionIntervention = "Autonomous roadside safety stop";
  1009→  1009→    this.mode = "AUTONOMOUS";
  1010→  1010→    this.controlLocked = true;
  1011→  1011→    this.hazardsOn = true;
  1012→  1012→    this.disengageAutopilot("autonomous safety takeover");
  1013→  1013→    this.firstCriticalAt ??= Date.now();
  1014→  1014→    if (!this.takeoverPlanned) {
  1015→  1015→      this.takeoverPlanned = true;
  1016→  1016→      this.sessionReason = this.decision.reason;
  1017→  1017→      planTakeover(
  1018→  1018→        this.autonomy,
  1019→  1019→        this.kin.speed,
  1020→  1020→        this.kin.position,
  1021→  1021→        this.kin.lateral,
  1022→  1022→        VEHICLE_SPECS[this.vehicleType],
  1023→  1023→        s.roadsideTarget,
  1024→  1024→      );
  1025→  1025→      this.emit(
  1026→  1026→        "AI",
  1027→  1027→        "SAFETY ENGINE",
  1028→  1028→        "AUTONOMOUS TAKEOVER — manual control disabled",
  1029→  1029→      );
  1030→  1030→      this.emit(
  1031→  1031→        "AI",
  1032→  1032→        "VEHICLE ENGINE",
  1033→  1033→        `Safe stop zone designated at +${Math.round(this.autonomy.stopZone.position - this.kin.position)} m`,
  1034→  1034→      );
  1035→  1035→      playCue("TAKEOVER");
  1036→  1036→    }
  1037→  1037→  }

  1042→  1042→    decision: SafetyDecision,
  1043→  1043→  ) {

  1050→  1050→    } else if (next === "WARNING") {

  1057→  1057→          ? "WARNING"
  1058→  1058→          : "CRITICAL",
  1059→  1059→      "SAFETY ENGINE",
  1060→  1060→      decision.reason,
  1061→  1061→    );

  1083→  1083→  emit(type: TimelineEvent["type"], source: TimelineEvent["source"], message: string) {
  1084→  1084→    // dedupe identical messages within 1.5 s
  1085→  1085→    const key = `${type}:${message}`;
  1086→  1086→    const last = this.lastEventAt.get(key) ?? -10;
  1087→  1087→    if (this.clock - last < 1.5) return;
  1088→  1088→    this.lastEventAt.set(key, this.clock);
  1089→  1089→
  1090→  1090→    if (type === "INFO") this.sessionEventTally.info += 1;
  1091→  1091→    if (type === "WARNING") this.sessionEventTally.warning += 1;
  1092→  1092→    if (type === "CRITICAL") this.sessionEventTally.critical += 1;
  1093→  1093→    if (type === "AI") this.sessionEventTally.ai += 1;
  1094→  1094→
  1095→  1095→    const e: TimelineEvent = {
  1096→  1096→      id: makeEventId(),
  1097→  1097→      ts: Date.now(),
  1098→  1098→      type,
  1099→  1099→      source,
  1100→  1100→      message,
  1101→  1101→    };
  1102→  1102→    this.listener?.onEvent(e);
  1103→  1103→  }

  1107→  1107→      ? (Date.now() - this.sessionStartedAt) / 1000

  1110→  1110→      this.firstCriticalAt && this.sessionStartedAt
  1111→  1111→        ? (Date.now() - this.firstCriticalAt) / 1000

  1119→  1119→      intervention: this.sessionIntervention,
  1120→  1120→      interventionTimeSec,
  1121→  1121→      reason:
  1122→  1122→        this.sessionReason ??
  1123→  1123→        (this.decision.trigger
  1124→  1124→          ? this.decision.reason
  1125→  1125→          : result === "MANUAL_STOP"
  1126→  1126→            ? "Operator requested stop"
  1127→  1127→            : "Routine stop"),
