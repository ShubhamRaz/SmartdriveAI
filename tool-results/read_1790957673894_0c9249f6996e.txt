   400→   400→  tick(rawDt: number) {
   401→   401→    const dt = Math.min(rawDt, 0.05);
   402→   402→    this.clock += dt;
   403→   403→    const s = this.settings;
   404→   404→
   405→   405→    // ---------- vision merge ----------
   406→   406→    this.updateDriverAndDrowsiness(dt, s);
   407→   407→
   408→   408→    // ---------- mode machine ----------
   409→   409→    switch (this.mode) {
   410→   410→      case "SAFETY_CHECK":
   411→   411→        this.tickSafetyCheck(dt);
   412→   412→        break;
   413→   413→      case "READY":
   414→   414→      case "MANUAL":
   415→   415→      case "WARNING":
   416→   416→        this.tickDriving(dt, s);
   417→   417→        break;
   418→   418→      case "AUTONOMOUS":
   419→   419→        this.tickAutonomous(dt);
   420→   420→        break;
   421→   421→      case "STOPPING":
   422→   422→        this.tickStopping(dt);
   423→   423→        break;
   424→   424→      case "EMERGENCY":
   425→   425→        this.tickEmergency(dt);
   426→   426→        break;
   427→   427→      case "STOPPED":
   428→   428→        this.kin.throttle = 0;
   429→   429→        this.kin.brake = 0;
   430→   430→        this.kin.steering = 0;
   431→   431→        break;
   432→   432→      case "IDLE":
   433→   433→        break;
   434→   434→    }
   435→   435→
   436→   436→    // ---------- safety evaluation (10 Hz) ----------
   437→   437→    this.safetyPulse(dt, s);
   438→   438→  }

   441→   441→
   442→   442→  private tickSafetyCheck(dt: number) {
   443→   443→    this.safetyCheckTimer += dt;
   444→   444→    if (!this.helmetCheckAnnounced && this.safetyCheckTimer > 0.5) {
   445→   445→      this.helmetCheckAnnounced = true;
   446→   446→      if (VEHICLE_RULES[this.vehicleType].requireHelmetToStart) {
   447→   447→        this.emit(
   448→   448→          this.helmet.state === "HELMET_DETECTED" ? "INFO" : "WARNING",
   449→   449→          "AI VISION",
   450→   450→          this.helmet.state === "HELMET_DETECTED"
   451→   451→            ? "Helmet detected (simulated detector)"
   452→   452→            : "Helmet NOT detected",
   453→   453→        );
   454→   454→      }
   455→   455→      this.emit(
   456→   456→        this.alcohol.detected ? "CRITICAL" : "INFO",
   457→   457→        "AI VISION",
   458→   458→        this.alcohol.detected
   459→   459→          ? `Alcohol sensor reading ${this.alcohol.level} — above threshold`
   460→   460→          : "Alcohol sensor reading 0 — within safe range",
   461→   461→      );
   462→   462→    }
   463→   463→    if (this.safetyCheckTimer >= 1.6) {
   464→   464→      // finalize checks
   465→   465→      const blockedHelmet =
   466→   466→        VEHICLE_RULES[this.vehicleType].requireHelmetToStart &&
   467→   467→        this.helmet.state !== "HELMET_DETECTED";
   468→   468→      const blockedAlcohol = this.alcohol.detected;
   469→   469→      if (blockedHelmet || blockedAlcohol) {
   470→   470→        this.mode = "IDLE";
   471→   471→        this.engineOn = false;
   472→   472→        this.startBlockedReason = blockedHelmet
   473→   473→          ? "HELMET NOT DETECTED"
   474→   474→          : "ALCOHOL DETECTED";
   475→   475→        this.emit(
   476→   476→          "CRITICAL",
   477→   477→          "SAFETY ENGINE",
   478→   478→          `START BLOCKED — ${this.startBlockedReason}`,
   479→   479→        );
   480→   480→        this.safetyStatus = "CRITICAL";
   481→   481→        this.decision = {
   482→   482→          severity: "CRITICAL",
   483→   483→          reason: blockedHelmet
   484→   484→            ? "Helmet interlock: no helmet detected"
   485→   485→            : "Alcohol interlock: sensor above threshold",
   486→   486→          recommendedAction: "BLOCK_START",
   487→   487→          controlMode: "LOCKED",
   488→   488→          evidence: [],
   489→   489→          trigger: blockedHelmet ? "HELMET" : "ALCOHOL",
   490→   490→        };
   491→   491→        this.sessionEventTally.critical += 1;
   492→   492→        this.prevSeverity = "CRITICAL";
   493→   493→        playCue("BLOCKED");
   494→   494→      } else {
   495→   495→        this.mode = "READY";
   496→   496→        this.engineOn = true;
   497→   497→        this.hazardsOn = false;
   498→   498→        this.sessionStartedAt = Date.now();
   499→   499→        this.emit("SAFE", "SAFETY ENGINE", "Safety checks passed");
   500→   500→        this.emit("INFO", "VEHICLE ENGINE", "Engine started — manual control available");
   501→   501→      }
   502→   502→    }
   503→   503→  }

   506→   506→    // manual / warning driving: gather inputs
   507→   507→    let throttle = this.kin.throttle;
   508→   508→    const brake = this.kin.brake;
   509→   509→    const steer = this.kin.steering;
   510→   510→
   511→   511→    if (this.demoAutoThrottle) {
   512→   512→      const cruise = (s.cruiseSpeed[this.vehicleType] / 3.6) * 0.92;
   513→   513→      throttle = this.kin.speed < cruise ? 1 : 0.05;
   514→   514→    }
   515→   515→
   516→   516→    const res = stepPhysics(
   517→   517→      this.kin.speed,
   518→   518→      this.kin.position,
   519→   519→      this.kin.lateral,
   520→   520→      this.kin.odometer,
   521→   521→      VEHICLE_SPECS[this.vehicleType],
   522→   522→      { throttle, brake, steer, powerLimit: this.powerLimit },
   523→   523→      dt,
   524→   524→      null,
   525→   525→    );
   526→   526→    this.applyPhysics(res);
   527→   527→
   528→   528→    // mode semantics
   529→   529→    if (this.decision.severity === "WARNING") {
   530→   530→      this.mode = "WARNING";
   531→   531→    } else {
   532→   532→      this.mode = this.kin.speed > 0.5 ? "MANUAL" : "READY";
   533→   533→    }
   534→   534→
   535→   535→    if (this.manualStopRequested) {
   536→   536→      this.manualStopRequested = false;
   537→   537→      this.beginControlledStop("MANUAL_STOP");
   538→   538→    }
   539→   539→  }

   548→   548→      dt,

   563→   563→      },
   564→   564→      dt,
   565→   565→      cmd.lateralTarget,
   566→   566→    );
   567→   567→    this.applyPhysics(res);
   568→   568→
   569→   569→    if (done) {
   570→   570→      this.finalizeStop("SAFE_STOP");
   571→   571→    }

   680→   680→
   681→   681→  private updateDriverAndDrowsiness(dt: number, s: AppSettings) {
   682→   682→    const visionActive = this.vision.active;
   683→   683→    const source: DetectionSource = this.simDrowsiness
   684→   684→      ? "SIMULATION"
   685→   685→      : visionActive
   686→   686→        ? "CAMERA"
   687→   687→        : "NONE";
   688→   688→
   689→   689→    const eyeClosed =
   690→   690→      this.simDrowsiness || (visionActive && s.drowsinessEnabled && this.vision.eyeClosed);
   691→   691→
   692→   692→    // temporal eye-closure accumulator
   693→   693→    if (eyeClosed) {
   694→   694→      this.drowsiness.closureDuration += dt;
   695→   695→      this.drowsiness.eyeClosed = true;
   696→   696→      this.drowsiness.source = source;
   697→   697→      const warn = Math.max(0.6, s.warningDuration);
   698→   698→      const crit = Math.max(warn + 0.4, s.criticalDuration);
   699→   699→      this.drowsiness.level =
   700→   700→        this.drowsiness.closureDuration >= crit
   701→   701→          ? "CRITICAL"
   702→   702→          : this.drowsiness.closureDuration >= warn
   703→   703→            ? "WARNING"
   704→   704→            : this.drowsiness.closureDuration > 0.18
   705→   705→              ? "BLINK"
   706→   706→              : "NORMAL";
   707→   707→      this.drowsiness.score = Math.min(
   708→   708→        100,
   709→   709→        (this.drowsiness.closureDuration / crit) * 100,
   710→   710→      );
   711→   711→      if (this.drowsiness.level === "WARNING" && !this.drowsinessWarnAnnounced) {
   712→   712→        this.drowsinessWarnAnnounced = true;
   713→   713→        this.emit(
   714→   714→          "WARNING",
   715→   715→          source === "CAMERA" ? "AI VISION" : "SIMULATION",
   716→   716→          `Drowsiness warning — eyes closed ${this.drowsiness.closureDuration.toFixed(1)} s`,
   717→   717→        );
   718→   718→        playCue("WARNING");
   719→   719→      }
   720→   720→      if (this.drowsiness.level === "CRITICAL" && !this.criticalAnnounced) {
   721→   721→        this.criticalAnnounced = true;
   722→   722→        this.emit(
   723→   723→          "CRITICAL",
   724→   724→          source === "CAMERA" ? "AI VISION" : "SIMULATION",
   725→   725→          `Critical drowsiness — eyes closed ${this.drowsiness.closureDuration.toFixed(1)} s, driver unresponsive`,
   726→   726→        );
   727→   727→        playCue("CRITICAL");
   728→   728→      }
   729→   729→    } else {
   730→   730→      if (this.drowsiness.closureDuration > 0.18) {
   731→   731→        this.drowsiness.blinkCount += 1;
   732→   732→      }
   733→   733→      const wasWarning =
   734→   734→        this.drowsiness.level === "WARNING" || this.drowsiness.level === "CRITICAL";
   735→   735→      if (wasWarning) {
   736→   736→        this.emit("INFO", "SYSTEM", "Driver alertness restored — drowsiness cleared");
   737→   737→      }
   738→   738→      this.drowsiness.closureDuration = 0;
   739→   739→      this.drowsiness.eyeClosed = false;
   740→   740→      this.drowsiness.level = "NORMAL";
   741→   741→      this.drowsiness.score = Math.max(0, this.drowsiness.score - dt * 60);
   742→   742→      this.drowsinessWarnAnnounced = false;
   743→   743→      this.criticalAnnounced = false;
   744→   744→      if (this.simDrowsiness) {
   745→   745→        // sim toggle on but treated as open (shouldn't happen) — keep closed flag
   746→   746→        this.drowsiness.eyeClosed = true;
   747→   747→      }
   748→   748→    }
   749→   749→    this.drowsiness.source = this.simDrowsiness
   750→   750→      ? "SIMULATION"
   751→   751→      : visionActive
   752→   752→        ? "CAMERA"
   753→   753→        : "NONE";
   754→   754→
   755→   755→    // driver presence
   756→   756→    if (visionActive && s.driverPresenceDetection) {
   757→   757→      this.driver.faceDetected = this.vision.faceDetected;
   758→   758→      this.driver.source = "CAMERA";
   759→   759→      if (!this.vision.faceDetected) {
   760→   760→        if (this.faceAbsentSince === null) this.faceAbsentSince = this.clock;
   761→   761→        if (
   762→   762→          !this.faceAbsentAnnounced &&
   763→   763→          this.clock - this.faceAbsentSince > s.faceAbsenceTimeout
   764→   764→        ) {
   765→   765→          this.faceAbsentAnnounced = true;
   766→   766→          this.emit("WARNING", "AI VISION", "Driver face not detected by camera");
   767→   767→        }
   768→   768→      } else {
   769→   769→        if (this.faceAbsentAnnounced) {
   770→   770→          this.emit("INFO", "AI VISION", "Driver face re-detected");
   771→   771→        }
   772→   772→        this.faceAbsentSince = null;
   773→   773→        this.faceAbsentAnnounced = false;
   774→   774→      }
   775→   775→    } else {
   776→   776→      this.driver.faceDetected = true;
   777→   777→      this.driver.source = "NONE";
   778→   778→      this.faceAbsentSince = null;
   779→   779→      this.faceAbsentAnnounced = false;
   780→   780→    }
   781→   781→    this.driver.present =
   782→   782→      this.driver.source !== "CAMERA" || this.driver.faceDetected;
   783→   783→    const attentionTarget =
   784→   784→      this.drowsiness.level === "CRITICAL"
   785→   785→        ? 12
   786→   786→        : this.drowsiness.level === "WARNING"
   787→   787→          ? 45
   788→   788→          : this.drowsiness.level === "BLINK"
   789→   789→            ? 88
   790→   790→            : !this.driver.present
   791→   791→              ? 55
   792→   792→              : 97;
   793→   793→    this.driver.attention +=
   794→   794→      (attentionTarget - this.driver.attention) * Math.min(1, dt * 3);
   795→   795→  }

   802→   802→  /* --------------------------- safety pulse ------------------------------ */
   803→   803→
   804→   804→  private safetyAccum = 0;
   805→   805→
   806→   806→  private safetyPulse(dt: number, s: AppSettings) {
   807→   807→    this.safetyAccum += dt;
   808→   808→    if (this.safetyAccum < 0.1) return;
   809→   809→    this.safetyAccum = 0;
   810→   810→
   811→   811→    const input = {
   812→   812→      vehicleType: this.vehicleType,
   813→   813→      vehicleMode: this.mode,
   814→   814→      helmet: this.helmet,
   815→   815→      alcohol: this.alcohol,
   816→   816→      alcoholEscalated: this.alcoholEscalated(),
   817→   817→      drowsiness: this.drowsiness,
   818→   818→      driver: this.driver,
   819→   819→      accident: this.accident,
   820→   820→      engineOn: this.engineOn,
   821→   821→      moving: this.kin.speed > 0.5,
   822→   822→    };
   823→   823→    const decision = evaluateSafety(input);
   824→   824→    this.decision = decision;
   825→   825→
   826→   826→    // severity transitions → events + audio
   827→   827→    if (decision.severity !== this.prevSeverity) {
   828→   828→      this.onSeverityChange(this.prevSeverity, decision.severity, decision);
   829→   829→      this.prevSeverity = decision.severity;
   830→   830→    }
   831→   831→
   832→   832→    // periodic warning beeps while in warning
   833→   833→    if (
   834→   834→      decision.severity === "WARNING" &&
   835→   835→      this.mode !== "IDLE" &&
   836→   836→      this.clock - this.lastWarnBeep > 2.2
   837→   837→    ) {
   838→   838→      this.lastWarnBeep = this.clock;
   839→   839→      playCue("WARNING");
   840→   840→    }
   841→   841→
   842→   842→    // apply decision actions to the state machine
   843→   843→    // EMERGENCY applies from any active mode (including autonomous)
   844→   844→    if (
   845→   845→      decision.recommendedAction === "EMERGENCY_STOP" &&
   846→   846→      this.mode !== "IDLE" &&
   847→   847→      this.mode !== "SAFETY_CHECK" &&
   848→   848→      this.mode !== "STOPPED"
   849→   849→    ) {
   850→   850→      this.sessionCounts.interventions += 1;
   851→   851→      this.sessionIntervention = "Emergency stop";
   852→   852→      this.enterEmergency("Accident / impact detected");
   853→   853→      return;
   854→   854→    }
   855→   855→
   856→   856→    if (this.mode === "READY" || this.mode === "MANUAL" || this.mode === "WARNING") {
   857→   857→      switch (decision.recommendedAction) {
   858→   858→        case "WARN":
   859→   859→          this.controlLocked = false;
   860→   860→          this.safetyStatus = "WARNING";
   861→   861→          break;
   862→   862→        case "REDUCE_POWER": {
   863→   863→          // bike: motor power reduced → glide to a controlled stop
   864→   864→          this.safetyStatus = "CRITICAL";
   865→   865→          this.sessionCounts.interventions += 1;
   866→   866→          this.sessionIntervention = "Motor power reduction + controlled stop";
   867→   867→          this.bikeMotorLimited = true;
   868→   868→          this.powerLimit = 0.25;
   869→   869→          this.emit(
   870→   870→            "WARNING",
   871→   871→            "SAFETY ENGINE",
   872→   872→            "Motor power reduced — controlled shutdown",
   873→   873→          );
   874→   874→          if (this.kin.speed > 0.2) {
   875→   875→            this.beginControlledStop("SAFE_STOP", null, 0.35);
   876→   876→          } else {
   877→   877→            this.finalizeStop("SAFE_STOP");
   878→   878→          }
   879→   879→          break;
   880→   880→        }
   881→   881→        case "CONTROLLED_STOP":
   882→   882→          this.safetyStatus = "CRITICAL";
   883→   883→          this.sessionCounts.interventions += 1;
   884→   884→          this.sessionIntervention = "Controlled safety stop";
   885→   885→          this.beginControlledStop("SAFE_STOP");
   886→   886→          break;
   887→   887→        case "AUTONOMOUS_STOP":
   888→   888→          this.safetyStatus = "CRITICAL";
   889→   889→          this.beginAutonomousStop(s);
   890→   890→          break;
   891→   891→        case "BLOCK_START":
   892→   892→          // engine running but interlock demands hold (e.g., alcohol while idle)
   893→   893→          this.safetyStatus = "CRITICAL";
   894→   894→          if (this.engineOn) {
   895→   895→            this.beginControlledStop("SAFE_STOP");
   896→   896→          }
   897→   897→          break;
   898→   898→        default:
   899→   899→          this.safetyStatus = this.mode === "STOPPED" ? this.safetyStatus : "SAFE";
   900→   900→      }
   901→   901→    } else if (
   902→   902→      this.mode === "IDLE" &&
   903→   903→      decision.recommendedAction === "BLOCK_START" &&
   904→   904→      decision.trigger === "HELMET"
   905→   905→    ) {
   906→   906→      // keep the interlock visible on the dashboard
   907→   907→      this.safetyStatus = "WARNING";
   908→   908→      this.startBlockedReason = this.startBlockedReason ?? "HELMET NOT DETECTED";
   909→   909→    }
   910→   910→  }

   946→   946→    prev: SafetyStatus,
   947→   947→    next: SafetyStatus,
   948→   948→    decision: SafetyDecision,
   949→   949→  ) {

   962→   962→        : next === "WARNING"
   963→   963→          ? "WARNING"
   964→   964→          : "CRITICAL",
   965→   965→      "SAFETY ENGINE",
   966→   966→      decision.reason,
   967→   967→    );

   971→   971→
   972→   972→  private applyPhysics(res: {
   973→   973→    speed: number;
   974→   974→    position: number;
   975→   975→    lateral: number;

   978→   978→  }) {
   979→   979→    this.kin.accel = (res.speed - this.kin.speed) / 0.016;

   988→   988→
   989→   989→  emit(type: TimelineEvent["type"], source: TimelineEvent["source"], message: string) {
   990→   990→    // dedupe identical messages within 1.5 s
   991→   991→    const key = `${type}:${message}`;
   992→   992→    const last = this.lastEventAt.get(key) ?? -10;
   993→   993→    if (this.clock - last < 1.5) return;
   994→   994→    this.lastEventAt.set(key, this.clock);
   995→   995→
   996→   996→    if (type === "INFO") this.sessionEventTally.info += 1;
   997→   997→    if (type === "WARNING") this.sessionEventTally.warning += 1;
   998→   998→    if (type === "CRITICAL") this.sessionEventTally.critical += 1;
   999→   999→    if (type === "AI") this.sessionEventTally.ai += 1;
  1000→  1000→
  1001→  1001→    const e: TimelineEvent = {
  1002→  1002→      id: makeEventId(),
  1003→  1003→      ts: Date.now(),
  1004→  1004→      type,
  1005→  1005→      source,
  1006→  1006→      message,
  1007→  1007→    };
  1008→  1008→    this.listener?.onEvent(e);
  1009→  1009→  }

  1011→  1011→  private emitFinalReport(result: SessionRecord["result"]) {
  1012→  1012→    const durationSec = this.sessionStartedAt
  1013→  1013→      ? (Date.now() - this.sessionStartedAt) / 1000
  1014→  1014→      : 0;
  1015→  1015→    const interventionTimeSec =
  1016→  1016→      this.firstCriticalAt && this.sessionStartedAt
  1017→  1017→        ? (Date.now() - this.firstCriticalAt) / 1000
  1018→  1018→        : null;
  1019→  1019→    const record: SessionRecord = {
  1020→  1020→      id: makeSessionId(),
  1021→  1021→      vehicle: this.vehicleType,
  1022→  1022→      startedAt: this.sessionStartedAt ?? Date.now(),
  1023→  1023→      durationSec,
  1024→  1024→      result,
  1025→  1025→      intervention: this.sessionIntervention,
  1026→  1026→      interventionTimeSec,
  1027→  1027→      reason:
  1028→  1028→        this.sessionReason ??
  1029→  1029→        (this.decision.trigger
  1030→  1030→          ? this.decision.reason
  1031→  1031→          : result === "MANUAL_STOP"
  1032→  1032→            ? "Operator requested stop"
  1033→  1033→            : "Routine stop"),
  1034→  1034→      eventCounts: { ...this.sessionEventTally },
